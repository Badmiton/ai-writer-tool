/** * 生成提示词 (非流式) - 完整调试版本 */ async function generatePrompt(formData) { const generateBtn = document.getElementById('generateBtn'); const originalText = generateBtn.innerHTML; generateBtn.innerHTML = '⏳AI生成中...'; generateBtn.disabled = true; document.getElementById('loading').classList.remove('hidden'); document.getElementById('promptResult').style.display = 'none'; try { console.log('=== 开始请求 ==='); console.log('API地址:', `${API_BASE_URL}/aigc-api/common/prompt/generate`); console.log('使用的API密钥:', formData.apiKey.substring(0, 10) + '...' + formData.apiKey.substring(formData.apiKey.length - 4)); console.log('请求数据:', { ...formData, apiKey: '***隐藏***', sampleContent: formData.sampleContent.substring(0, 50) + '...' }); const response = await fetch(`${API_BASE_URL}/aigc-api/common/prompt/generate`, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(formData) }); console.log('=== 响应状态 ==='); console.log('HTTP状态码:', response.status); console.log('HTTP状态文本:', response.statusText); console.log('响应头:', Object.fromEntries(response.headers.entries())); // 获取响应文本 const responseText = await response.text(); console.log('响应原始文本:', responseText); // 尝试解析JSON let data; try { data = JSON.parse(responseText); console.log('解析后的响应数据:', data); } catch (parseError) { throw new Error(`JSON解析失败: ${parseError.message}\n原始响应: ${responseText}`); } // 隐藏加载动画 document.getElementById('loading').classList.add('hidden'); document.getElementById('promptResult').style.display = 'block'; // 恢复按钮状态 generateBtn.innerHTML = '🚀生成AI写作提示词'; generateBtn.disabled = false; if (data.success) { document.getElementById('promptResult').textContent = data.data.prompt; if (data.data.authorName && !formData.authorName) { document.getElementById('authorName').value = data.data.authorName; } } else { // 显示超详细的错误信息 const errorMsg = `❌ 生成失败\n\n` + `【错误详情】\n` + `错误代码: ${data.code || '未知'}\n` + `错误信息: ${data.message || data.error || '未知错误'}\n` + `HTTP状态: ${response.status} ${response.statusText}\n\n` + `【请求信息】\n` + `API密钥前缀: ${formData.apiKey.substring(0, 10)}...\n` + `API密钥后缀: ...${formData.apiKey.substring(formData.apiKey.length - 4)}\n\n` + `【完整错误对象】\n` + JSON.stringify(data, null, 2); document.getElementById('promptResult').textContent = errorMsg; console.error('❌ API返回错误:', data); } } catch (error) { console.error('❌ 请求异常:', error); document.getElementById('loading').classList.add('hidden'); document.getElementById('promptResult').style.display = 'block'; const errorMsg = `❌ 请求失败\n\n` + `【错误信息】\n` + `${error.message}\n\n` + `【错误堆栈】\n` + `${error.stack || '无堆栈信息'}\n\n` + `请打开浏览器控制台(F12)查看详细日志`; document.getElementById('promptResult').textContent = errorMsg; generateBtn.innerHTML = '🚀生成AI写作提示词'; generateBtn.disabled = false; } }
